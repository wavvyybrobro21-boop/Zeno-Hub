-- Info Tab
local InfoTab = Window:Tab({
    Name = "Info",
    Icon = "rbxassetid://13524671408"
})

local PlayerInfoSection = InfoTab:Section("Player Info")
PlayerInfoSection:Label("Username: " .. localPlayer.Name)

local ViewportSection = InfoTab:Section("Character View")
local ViewportFrame = ViewportSection:Viewport(function(viewport)
    local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local clonedChar = character:Clone()
    clonedChar.Parent = viewport

    local camera = Instance.new("Camera", viewport)
    viewport.CurrentCamera = camera

    local hrp = clonedChar:WaitForChild("HumanoidRootPart")
    if hrp then
        camera.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 5, 10)) * CFrame.new(hrp.Position)
    end
end)

local FlySection = PlayerTab:Section("Fly")

local isFlying = false
local flySpeed = 5
local playerMovement = nil

FlySection:Toggle({
    Name = "Fly",
    Callback = function(value)
        isFlying = value
        if isFlying then
            playerMovement = localPlayer.Character:FindFirstChild("HumanoidRootPart")
            if playerMovement then
                local originalGravity = Workspace.Gravity
                Workspace.Gravity = 0
                
                local newRootPart = Instance.new("Part")
                newRootPart.Name = "FlyRootPart"
                newRootPart.Size = Vector3.new(2, 2, 2)
                newRootPart.Anchored = true
                newRootPart.CanCollide = false
                newRootPart.Parent = localPlayer.Character
                
                local flySpeedValue = 5
                
                local function updateFly()
                    if not isFlying or not localPlayer.Character then return end
                    
                    local moveDirection = Vector3.new(0, 0, 0)
                    if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDirection += Vector3.new(0, 0, -1) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDirection += Vector3.new(0, 0, 1) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDirection += Vector3.new(-1, 0, 0) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDirection += Vector3.new(1, 0, 0) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDirection += Vector3.new(0, 1, 0) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDirection += Vector3.new(0, -1, 0) end
                    
                    moveDirection = moveDirection.Unit * flySpeedValue
                    newRootPart.CFrame = newRootPart.CFrame + moveDirection
                end
                
                RunService.RenderStepped:Connect(updateFly)
            end
        else
            Workspace.Gravity = 196.2
            if localPlayer.Character then
                local flyRootPart = localPlayer.Character:FindFirstChild("FlyRootPart")
                if flyRootPart then
                    flyRootPart:Destroy()
                end
            end
        end
    end
})

FlySection:Slider({
    Name = "Fly Speed",
    Min = 1,
    Max = 50,
    Default = 5,
    Callback = function(value)
        flySpeed = value
    end
})

local function applyMovementModifications()
    if localPlayer.Character then
        local humanoid = localPlayer.Character:FindFirstChild("Humanoid")
        if humanoid then
            humanoid.WalkSpeed = 16 * speedValue
            humanoid.JumpPower = 50 * jumpValue
        end
    end
end

local function onCharacterAdded(character)
    applyMovementModifications()
end

localPlayer.CharacterAdded:Connect(onCharacterAdded)
applyMovementModifications()

local PlayerTeleportSection = TeleportTab:Section("Teleport to Player")

local selectedPlayer = nil
local playerList = {}

local function updatePlayerList()
    playerList = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            table.insert(playerList, player.Name)
        end
    end
end

updatePlayerList()

local playerDropdown = PlayerTeleportSection:Dropdown({
    Name = "Select Player",
    Options = playerList,
    Callback = function(value)
        selectedPlayer = value
    end
})

Players.PlayerAdded:Connect(function(player)
    updatePlayerList()
    playerDropdown:SetOptions(playerList)
end)

Players.PlayerRemoving:Connect(function(player)
    updatePlayerList()
    playerDropdown:SetOptions(playerList)
end)

PlayerTeleportSection:Button({
    Name = "Teleport to Selected Player",
    Callback = function()
        if selectedPlayer then
            local targetPlayer = Players:FindFirstChild(selectedPlayer)
            if targetPlayer and targetPlayer.Character then
                local targetRootPart = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
                if targetRootPart then
                    if localPlayer.Character then
                        local playerRootPart = localPlayer.Character:FindFirstChild("HumanoidRootPart")
                        if playerRootPart then
                            playerRootPart.CFrame = targetRootPart.CFrame + Vector3.new(0, 5, 0)
                            Library:Notify({
                                Title = "Teleported",
                                Content = "Teleported to " .. selectedPlayer,
                                Time = 3
                            })
                        end
                    end
                end
            end
        else
            Library:Notify({
                Title = "Error",
                Content = "Please select a player first",
                Time = 3
            })
        end
    end
})

-- Autofarm Tab
local AutofarmTab = Window:Tab({
    Name = "Autofarm",
    Icon = "rbxassetid://13524671408"
})

local FarmSection = AutofarmTab:Section("Farm Settings")

local isFarming = false
local farmLoop = nil

FarmSection:Toggle({
    Name = "Enable Autofarm",
    Callback = function(value)
        isFarming = value
        if isFarming then
            farmLoop = RunService.Heartbeat:Connect(function()
                if localPlayer.Character then
                    local humanoidRootPart = localPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if humanoidRootPart then
                        local randomX = math.random(-50, 50)
                        local randomZ = math.random(-50, 50)
                        local targetPosition = Vector3.new(randomX, humanoidRootPart.Position.Y, randomZ)
                        humanoidRootPart.CFrame = CFrame.new(targetPosition)
                    end
                end
            end)
        else
            if farmLoop then
                farmLoop:Disconnect()
            end
        end
    end
})
